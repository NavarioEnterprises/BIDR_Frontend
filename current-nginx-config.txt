# NGINX reverse proxy for BIDR services
upstream auth_service {
    server auth-service.bidr-uat.svc.cluster.local:8001;
}
upstream chat_service {
    server chat-service.bidr-uat.svc.cluster.local:8002;
}
upstream payment_service {
    server payment-service.bidr-uat.svc.cluster.local:8003;
}
upstream product_management_service {
    server product-management-service.bidr-uat.svc.cluster.local:8004;
}
upstream notifications_service {
    server notifications-service.bidr-uat.svc.cluster.local:8005;
}
upstream transactions_service {
    server transactions-service.bidr-uat.svc.cluster.local:8006;
}
upstream reviews_service {
    server reviews-service.bidr-uat.svc.cluster.local:8007;
}
upstream resolution_service {
    server resolution-service.bidr-uat.svc.cluster.local:8008;
}

server {
    listen 80 default_server;
    server_name _;

    # Health
    location /health {
        access_log off;
        return 200 "ok\n";
        add_header Content-Type text/plain;
    }

    # Auth service
    location /api/auth/ {
        proxy_pass http://auth_service/api/auth/;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
    }
    location /auth/ {
        proxy_pass http://auth_service/auth/;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
    }

    # Chat service - complete isolation
    location ^~ /chat/ {
        # Rewrite /chat/xxx -> /xxx for the backend
        rewrite ^/chat/(.*)$ /$1 break;
        proxy_pass http://chat_service;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        # Essential: rewrite ALL admin URLs to include /chat prefix
        proxy_redirect ~^/admin(.*)$ /chat/admin$1;
        proxy_redirect ~^/accounts(.*)$ /chat/accounts$1;
        proxy_redirect ~^/$ /chat/;
    }

    # Payment service - complete isolation  
    location ^~ /payments/ {
        rewrite ^/payments/(.*)$ /$1 break;
        proxy_pass http://payment_service;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        proxy_redirect ~^/admin(.*)$ /payments/admin$1;
        proxy_redirect ~^/accounts(.*)$ /payments/accounts$1;
        proxy_redirect ~^/$ /payments/;
    }

    # Product service - complete isolation
    location ^~ /products/ {
        rewrite ^/products/(.*)$ /$1 break;
        proxy_pass http://product_management_service;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        proxy_redirect ~^/admin(.*)$ /products/admin$1;
        proxy_redirect ~^/accounts(.*)$ /products/accounts$1;
        proxy_redirect ~^/$ /products/;
    }

    # Notifications service - complete isolation
    location ^~ /notifications/ {
        rewrite ^/notifications/(.*)$ /$1 break;
        proxy_pass http://notifications_service;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        proxy_redirect ~^/admin(.*)$ /notifications/admin$1;
        proxy_redirect ~^/accounts(.*)$ /notifications/accounts$1;
        proxy_redirect ~^/$ /notifications/;
    }

    # Transactions service - complete isolation
    location ^~ /transactions/ {
        rewrite ^/transactions/(.*)$ /$1 break;
        proxy_pass http://transactions_service;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        proxy_redirect ~^/admin(.*)$ /transactions/admin$1;
        proxy_redirect ~^/accounts(.*)$ /transactions/accounts$1;
        proxy_redirect ~^/$ /transactions/;
    }

    # Reviews service - complete isolation
    location ^~ /reviews/ {
        rewrite ^/reviews/(.*)$ /$1 break;
        proxy_pass http://reviews_service;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        proxy_redirect ~^/admin(.*)$ /reviews/admin$1;
        proxy_redirect ~^/accounts(.*)$ /reviews/accounts$1;
        proxy_redirect ~^/$ /reviews/;
    }

    # Resolution service - complete isolation
    location ^~ /resolution/ {
        rewrite ^/resolution/(.*)$ /$1 break;
        proxy_pass http://resolution_service;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        proxy_redirect ~^/admin(.*)$ /resolution/admin$1;
        proxy_redirect ~^/accounts(.*)$ /resolution/accounts$1;
        proxy_redirect ~^/$ /resolution/;
    }

    # Default: send to auth service
    location / {
        proxy_pass http://auth_service/;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
    }
}
